cmake_minimum_required(VERSION 3.24)
project(gst_sdl3_linux LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Build this prototype on Linux")
endif()
find_package(SDL3 3.4 CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-app-1.0>=1.24 gstreamer-video-1.0 gstreamer-gl-1.0 gstreamer-gl-egl-1.0)
pkg_check_modules(DMA REQUIRED IMPORTED_TARGET gstreamer-allocators-1.0 egl glesv2)
add_executable(gst_sdl3_linux main.cpp inspect.cpp external.cpp)
target_compile_features(gst_sdl3_linux PRIVATE cxx_std_17)
target_compile_options(gst_sdl3_linux PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(gst_sdl3_linux PRIVATE SDL3::SDL3 PkgConfig::GST PkgConfig::DMA)
